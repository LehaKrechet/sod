cmake_minimum_required(VERSION 3.16)
project(sod_multicast LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(sod_multicast src/main.cpp src/config.cpp src/network.cpp src/discovery.cpp)
target_compile_options(sod_multicast PRIVATE -O3 -Wall -Wextra -Wpedantic)
target_link_libraries(sod_multicast pthread)
